Language Tour

Learn Lambda Script

A guided tour of the language — from basic values to advanced document processing.

Data Types

Lambda Script has a rich set of primitive and composite types:

Primitives

42              // int
3.14            // float (64-bit)
123.45m         // decimal (exact)
42n             // integer (arbitrary precision)
"hello"         // string (UTF-8)
'symbol'        // symbol (interned identifier)
true            // bool
null            // null
t'2025-04-01'   // datetime
b'\xDEADBEEF'   // binary

Collections

let a = [1, 2, 3]                     // array
let m = {name: "Alice", age: 30}      // map
let e = <div class: "card", "text">   // element (markup literal)
let r = 1 to 10                       // range (inclusive)

Indexing & Slicing

let arr = [10, 20, 30, 40]
let user = {name: "Alice", age: 30}

arr[0]            // 10 — first element
arr[1 to 3]       // [20, 30, 40] — slice
user.name         // "Alice" — field access
"hello"[1 to 3]   // "ell" — string slicing

Variables & Bindings

// Immutable bindings (default)
let x = 42
let lang: string = "Lambda"

// Let expressions (inline bindings)
let three = (let a = 1, let b = 2, a + b)   // 3

// Mutable variables (in procedures)
pn main() {
    var n = 0
    n = n + 1
    print(n)    // 1
}

Pipe Operator & Data Pipelines

The pipe operators are Lambda's signature feature — fluent, set-oriented data transformations. |> maps, |: filters, and ~ is the current item:

let nums = [1, 2, 3, 4, 5]
let users = [{name: "Alice", age: 30}, {name: "Bob", age: 15}, {name: "Carol", age: 42}]

// Map with |>: apply a transformation to every item
nums |> ~ * 2                          // [2, 4, 6, 8, 10]

// Extract fields
users |> ~.name                        // ["Alice", "Bob", "Carol"]

// Filter with |:
nums |: ~ > 3                          // [4, 5]

// Chain operations: filter → map → aggregate
users |: ~.age >= 18 |> ~.name |> len  // 2 adults

// Spread a pipe result into an array literal
let spread = [0, *(nums |> ~ * 10), 99]  // [0, 10, 20, 30, 40, 50, 99]

Vector Arithmetic

Scalar operations automatically broadcast over collections:

let v = [1, 2]

1 + v               // [2, 3]
v * 2               // [2, 4]
v + [3, 4]          // [4, 6]
v ** 2              // [1, 4]

For-Expressions

SQL-like comprehensions with where, order by, limit, offset:

let data = [3, -1, 4, -5, 9]
let items = [{name: "pen", price: 2}, {name: "lamp", price: 30}, {name: "book", price: 12}]
let xs = [1, 5, 8], ys = [4, 6]

// Filter and transform
for (x in data where x > 0) x * 2                      // [6, 8, 18]

// With local bindings
for (x in data, let sq = x * x where sq > 10) sq       // [16, 25, 81]

// Sorting and pagination
for (x in items order by x.price desc limit 2) x.name  // ["lamp", "book"]

// Multi-source comprehension
for (x in xs, y in ys where x + y > 10) {x: x, y: y}   // (5, 6), (8, 4), (8, 6)

Type System

Lambda has a rich type system with inference and explicit annotations, similar to and beyond TypeScript:

// Type annotations
let x: int = 42
let items: string[] = ["a", "b"]

// Union and optional types
type Result = int | error
type Name = string?          // string | null

// Type declarations
type User = {name: string, age: int}
type HttpMethod = "GET" | "POST" | "PUT" | "DELETE"

Object Types

type Point {
    x: float, y: float,
    fn distance(other: Point) =>
        math.sqrt((x - other.x)**2 + (y - other.y)**2)
}

type Circle : Point { radius: float }   // inheritance

let p = <Point x: 3.0, y: 4.0>          // object literal
p.distance(<Point x: 0.0, y: 0.0>)      // 5 — method call
p is Point                              // true (nominal)
{x: 3.0, y: 4.0} is Point               // false — a plain map is not a Point

Element Type Patterns

// Schema types for document structures
type Section = <section string*>
type Link = <a href: string, string>
type Article = <article title: string, string, Section*>
type ImgTag = <img src: string, alt: string>

Functions

// Pure functions (fn)
fn add(a: int, b: int) => a + b
fn greet(who: string) => "Hello, " ++ who ++ "!"

// Anonymous functions
let double = (x) => x * 2
let add3 = (a, b, c) => a + b + c

// Procedures with mutation (pn)
pn main() {
    var total = 0
    for i in 0 to 9 {
        total = total + i
    }
    print(total)    // 45
}

Pattern Matching

fn describe(x) => match x {
    case null:             "nothing"
    case 0:                "zero"
    case 1 to 9:           "small number"      // range
    case int that (~ > 9): "big number"        // type + constraint
    case string:           "text: " ++ ~        // type
    case int[]:            "int array"          // collection type
    default:               "something else"
}

Query Operator

The ? query operator searches data trees by type or structure:

let html = input("page.html", 'html')^
let data = input("status.json", 'json')^

html?<img>                    // all <img> descendants
html?<div class: string>      // <div>s with a class attribute
data?{status: "ok"}           // maps where status == "ok"
html?<div>?<a>                // <a>s nested inside <div>s

Error Handling

// Functions that can fail declare T^E: a value, or a raised error
fn divide(a: float, b: float) float^ {
    if (b == 0) raise error("division by zero")
    else a / b
}

// Propagate with postfix ^: an error returns to the caller
fn compute(x: float) float^ {
    let r = divide(100, x)^
    r * 2
}

// Handle it with a braced handler, where ^ is the error
pn main() {
    let result = compute(0) ^ {
        print("Error: " ++ ^.message ++ "\n")
        0
    }
    print("Result: " ++ string(result) ++ "\n")
}

Document Processing

Lambda's signature capability — treat documents as structured data:

Markup Literals

let page = <html
    <head
        <title "My Page">
        <style "body { font-family: sans-serif; }">
    >
    <body
        <h1 "Welcome">
        <p "Lambda treats " <strong "documents"> " as data.">
    >
>

format(page, 'html')    // render to HTML string

Multi-Format Parsing

let md   = input("readme.md",    'markdown')^
let html = input("page.html",    'html')^
let json = input("data.json",    'json')^
let yaml = input("config.yaml",  'yaml')^
let csv  = input("data.csv",     'csv')^
let tex  = input("paper.tex",    'latex')^

Format Conversion

let data = {name: "Lambda", tags: ["fast", "functional"]}

// Direct format conversion
format(data, 'json')       // to a JSON string
format(data, 'yaml')       // to a YAML string
format(data, 'xml')        // to an XML string
