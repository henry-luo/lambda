% Lua erzeugt Pseudozufallszahlen und rechnet
\begin{luacode*}
math.randomseed(os.time())
function addaufgabe(hi)
tex.print{[[\begin{array}{rcrcr}]]}
for k = 1, hi do	
    a = math.random(11,50) 
	b = math.random(11,50)
    tex.print{a, '&+&', b, '&=&', a+b, '\\\\'}
	end
tex.print{[[\end{array}]]}
end    
function subaufgabe(hi)
tex.print{[[\begin{array}{rcrcr}]]}
for k = 1, hi do	
    a = math.random(11,50) 
	b = math.random(11,50)
    tex.print{a+b, '&-&', b, '&=&', a, '\\\\'}
	end
tex.print{[[\end{array}]]}
end   
function multaufgabe(hi)
tex.print{[[\begin{array}{rcrcr}]]}
for k = 1, hi do	
    a = math.random(5,20) 
	b = math.random(5,20)
	tex.print{a, '&\\cdot&', b, '&=&', a*b, '\\\\'}
    end
tex.print{[[\end{array}]]}
end    
function divaufgabe(hi)
tex.print{[[\begin{array}{rcrcr}]]}
for k = 1, hi do	
    a = math.random(5,20) 
	b = math.random(2,20)
    tex.print{a*b, '&:&', b, '&=&', a, '\\\\'}
	end
tex.print{[[\end{array}]]}
end   
\end{luacode*}
