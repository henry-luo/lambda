  printWidth: 80,
  semi: true,
  singleQuote: false,

function cloneOptions(options = {}) {
  return { ...DEFAULT_OPTIONS, ...options };
}
function normalizeText(value) {
  return value.trim().replace(/\s+/g, " ");
}
function isVisible(node) {
  return node != null && node.hidden !== true;
}
function renderInline(node, index) {
  const text = normalizeText(node.text || "");
  return { kind: "text", index, text, marks: node.marks || [] };
}
function renderBlock(node, index, context) {
  if (!isVisible(node)) return null;
  if (node.type === "heading")
    return {
      ...renderInline(node, index),
      level: node.level || 1,
      kind: "heading",
    };
  if (node.type === "paragraph")
    return { ...renderInline(node, index), kind: "paragraph" };
  if (node.type === "list")
    return {
      kind: "list",
      items: node.items.map((item, itemIndex) =>
        renderInline({ text: item }, itemIndex),
      ),
      context,
    };
  return { kind: "unknown", value: node };
}
function renderDocument(document, options) {
  const config = cloneOptions(options);
  const result = [];
  for (const [index, node] of document.blocks.entries()) {
    const rendered = renderBlock(node, index, { config });
    if (rendered !== null) result.push(rendered);
  }
  return { title: document.title, config, blocks: result };
}
class SourcePrinter {
  constructor(options = {}) {
    this.options = cloneOptions(options);
  }
  print(document) {
    const output = renderDocument(document, this.options);
    const lines = [];
    for (const [index, block] of output.blocks.entries()) {
      lines.push(
        index +
          1 +
          ". " +
          block.kind +
          ": " +
          (block.text || block.value || ""),
      );
    }
    return lines.join("\n");
  }
}
function formatDocument(document, options = {}) {
  return new SourcePrinter(options).print(document);
}

function collectNodes(
  root,
  options = {
    mode: "full",
    limit: 64,
    features: { metrics: true, paths: true },
  },
) {
  const entries = [];
  for (const branch of root.branches) {
    if (branch.enabled && branch.payload) {
      entries.push({
        id: branch.id,
        metrics: {
          count: branch.payload.metrics.count,
          weight: branch.payload.metrics.weight,
        },
        tags: branch.payload.tags,
        children: branch.payload.children,
        options,
      });
    } else {
      entries.push({ id: branch.id, skipped: true, reason: "disabled" });
    }
  }
  return { root, entries, options };
}
function scoreRecord(record) {
  let score = record.metrics.count * 2 + record.metrics.weight;
  if (record.tags.length > 2) {
    score = score + record.tags.length;
  }
  if (record.children.length > 1) {
    score = score + record.children[1].values[0];
  }
  return {
    id: record.id,
    score,
    detail: { tags: record.tags, first: record.children[0] },
  };
}
function walkLayers(tree) {
  const path = tree.meta.path;
  const terminal = tree.layer0;
  return { path, terminal, flags: tree.meta.flags };
}
function scoreEntry(entry) {
  if (entry.skipped) {
    return { id: entry.id, score: 0, detail: { reason: entry.reason } };
  }
  return scoreRecord(entry);
}
class Archive {
  constructor(snapshot) {
    this.snapshot = snapshot;
    this.entries = snapshot.entries;
  }
  describe() {
    return {
      name: this.snapshot.root.meta.name,
      count: this.entries.length,
      first: this.entries[0],
      terminal: this.snapshot.root.layer0,
    };
  }
  append(entry) {
    this.entries = this.entries.concat([entry]);
    return this.entries.length;
  }
}
